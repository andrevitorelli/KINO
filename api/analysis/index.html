    kino.analysis¶

    Ensemble analysis: moment maps, the Fisher-matrix forecast, and injection-recovery. See Forecasting a target for the forecast workflow.

    Ensemble analysis.

    Shape-noise characterisation, multiplicative/additive (m, c) bias calibration, instrument forecasts, and plotting over galaxy ensembles.

    kino.analysis.injection is deliberately not re-exported here: it depends on kino.inference, which itself depends on kino.analysis.moments, so importing it eagerly from this package would create an import cycle. Import it as from kino.analysis.injection import run_injection, summarize_trials instead.

    def compute_moment_maps(
    cube: Datacube,
    *,
    line_wavelength: u.Quantity | None = None,
    redshift: float | None = None,
    lsf_sigma: u.Quantity | None = None,
    snr_threshold: float = 3.0,
    subtract_background: bool | np.ndarray = True,
    check_window: bool = True,
) -> MomentMaps:
    """Flux/Vlos/dispersion moment maps from ``cube``.

    Parameters
    ----------
    cube : Datacube
    line_wavelength : Quantity (wavelength), optional
        Rest-frame line wavelength -- required if ``cube.spectral_kind ==
        "wavelength"`` (ignored, and unnecessary, for ``"velocity"``, where
        ``cube.spectral_axis`` already *is* a velocity).
    redshift : float or None, optional
        Overrides ``cube.redshift`` for the wavelength -> velocity
        conversion; ``None`` uses ``cube.redshift``.
    lsf_sigma : Quantity (velocity) or None, optional
        Instrumental broadening to remove from ``dispersion`` in quadrature
        (``sqrt(max(measured**2 - lsf_sigma**2, 0))``). ``None`` skips the
        correction, i.e. ``dispersion`` is the *observed* (instrument +
        intrinsic) width.
    snr_threshold : float, optional
        Spaxels with flux significance below this are masked (``vlos``,
        ``dispersion`` -> NaN). Only applied if ``cube.variance`` is set.
    subtract_background : bool or array_like, optional
        ``True`` (default): subtract a per-spaxel background estimated as the
        median flux across the spectral axis (appropriate for a roughly flat
        sky/dark pedestal occupying most of the band). **This estimator is
        fragile**: if the line fills much more than a third of the spectral
        window the median sits inside the line's wings, which biases the
        moments (``dispersion`` above all) and adds noise that the propagated
        errors do not include -- measured: ``dispersion_err`` 4.6x too small
        when the line filled ~half of a 20-channel window. ``False``: the cube
        is already background-subtracted. An **array** (broadcastable to
        ``cube.flux``, e.g. shape ``(n_spec, 1, 1)``): a *known* baseline to
        subtract -- what a forward model has, e.g.
        :meth:`kino.instruments.ifu.IFU.baseline_electrons`. Nothing is
        estimated from the data, so the errors stay calibrated at any window
        width; this is what ``IFUMomentLikelihood`` and the forecast use.
    check_window : bool, optional
        Run the cheap, cube-level too-wide-window diagnostic described in
        this module's docstring and emit a ``UserWarning`` if triggered.
        Set ``False`` to silence it -- e.g. when deliberately exploring a
        bad window (as `notebooks/07_moment_window_width_caveat.ipynb`
        does), or once you've already sized the window with
        :func:`crop_to_line` and don't need to be told twice.

    Returns
    -------
    MomentMaps
    """
    if check_window:
        fraction, n_eff, n_total = _effective_channel_fraction(cube.flux)
        if fraction < _MIN_EFFECTIVE_CHANNEL_FRACTION:
            warnings.warn(
                f"compute_moment_maps: only {fraction:.1%} of the {n_total} spectral "
                f"channels ({n_eff:.0f} effective) contribute to the stacked spectrum's "
                "flux -- the spectral window may be too wide relative to the line. "
                "dispersion is especially vulnerable to noise from the rest (see this "
                "module's 'window width matters' docstring section). Consider "
                "kino.analysis.moments.crop_to_line() first, or pass check_window=False "
                "to silence this.",
                UserWarning,
                stacklevel=2,
            )

    if cube.spectral_kind == "velocity":
        velocity = cube.spectral_axis.to(u.km / u.s)
    else:
        if line_wavelength is None:
            raise ValueError("line_wavelength is required for a wavelength-kind Datacube")
        z = cube.redshift if redshift is None else redshift
        line_center = line_wavelength * (1.0 + z)
        velocity = (C_LIGHT * (cube.spectral_axis / line_center - 1.0)).to(u.km / u.s)
    v = velocity.to_value(u.km / u.s)[:, np.newaxis, np.newaxis]

    flux_cube = cube.flux
    if isinstance(subtract_background, bool | np.bool_):
        background = np.median(flux_cube, axis=0, keepdims=True) if subtract_background else 0.0
    else:
        background = np.asarray(subtract_background, dtype=float)
        try:
            np.broadcast_shapes(background.shape, flux_cube.shape)
        except ValueError as exc:
            raise ValueError(
                f"known background of shape {background.shape} does not broadcast "
                f"to the cube {flux_cube.shape}"
            ) from exc
    signal = flux_cube - background

    with np.errstate(invalid="ignore", divide="ignore"):
        flux_map = signal.sum(axis=0)
        vlos_map = (signal * v).sum(axis=0) / flux_map
        dv = v - vlos_map[np.newaxis, :, :]
        sigma2_map = (signal * dv**2).sum(axis=0) / flux_map
    dispersion_map = np.sqrt(np.clip(sigma2_map, 0.0, None))

    if lsf_sigma is not None:
        lsf_kms = lsf_sigma.to_value(u.km / u.s)
        dispersion_map = np.sqrt(np.clip(dispersion_map**2 - lsf_kms**2, 0.0, None))

    flux_err = vlos_err = dispersion_err = snr = mask = None
    if cube.variance is not None:
        var = cube.variance
        var_d = var.sum(axis=0)
        flux_err = np.sqrt(var_d)

        var_n = (v**2 * var).sum(axis=0)
        cov_nd = (v * var).sum(axis=0)
        n_num = (signal * v).sum(axis=0)
        with np.errstate(invalid="ignore", divide="ignore"):
            var_vlos = (
                var_n / flux_map**2
                - 2.0 * n_num * cov_nd / flux_map**3
                + n_num**2 * var_d / flux_map**4
            )
        vlos_err = np.sqrt(np.clip(var_vlos, 0.0, None))

        var_m = (dv**4 * var).sum(axis=0)
        cov_md = (dv**2 * var).sum(axis=0)
        m_num = (signal * dv**2).sum(axis=0)
        with np.errstate(invalid="ignore", divide="ignore"):
            var_sigma2 = (
                var_m / flux_map**2
                - 2.0 * m_num * cov_md / flux_map**3
                + m_num**2 * var_d / flux_map**4
            )
            var_sigma = np.where(sigma2_map > 0, var_sigma2 / (4.0 * sigma2_map), np.nan)
        dispersion_err = np.sqrt(np.clip(var_sigma, 0.0, None))

        with np.errstate(invalid="ignore", divide="ignore"):
            snr = flux_map / flux_err
        mask = ~(snr >= snr_threshold)  # NaN-safe: NaN snr also masks out
        vlos_map = np.where(mask, np.nan, vlos_map)
        dispersion_map = np.where(mask, np.nan, dispersion_map)
        vlos_err = np.where(mask, np.nan, vlos_err)
        dispersion_err = np.where(mask, np.nan, dispersion_err)

    return MomentMaps(
        flux=flux_map,
        vlos=vlos_map,
        dispersion=dispersion_map,
        flux_err=flux_err,
        vlos_err=vlos_err,
        dispersion_err=dispersion_err,
        snr=snr,
        mask=mask,
        flux_unit=cube.flux_unit,
    )

    def crop_to_line(cube: Datacube, center: u.Quantity, half_width: u.Quantity) -> Datacube:
    """Crop ``cube``'s spectral axis to ``[center - half_width, center + half_width]``.

    The safe-path counterpart to this module's window-width caveat: size
    ``half_width`` to a handful of the line's expected total extent (Doppler
    range across the field, plus a few line widths of margin) and crop
    before calling :func:`compute_moment_maps`, rather than handing it an
    instrument's full bandpass.

    Parameters
    ----------
    cube : Datacube
    center : Quantity
        Line centre, in the same domain as ``cube.spectral_axis`` (a
        wavelength for ``spectral_kind="wavelength"``, a velocity for
        ``"velocity"``).
    half_width : Quantity
        Half-width of the retained window, same domain as ``center``.

    Returns
    -------
    Datacube
        A new ``Datacube`` with ``flux``/``variance``/``mask`` sliced to
        match the cropped ``spectral_axis``. Assumes ``cube.spectral_axis``
        is monotonically increasing (true of every cube produced elsewhere
        in kino).
    """
    axis = cube.spectral_axis
    lo, hi = center - half_width, center + half_width
    in_window = (axis >= lo) & (axis <= hi)
    if not np.any(in_window):
        raise ValueError(
            f"no channels of cube.spectral_axis (range {axis.min()}-{axis.max()}) "
            f"fall within [{lo}, {hi}]"
        )
    idx = np.flatnonzero(in_window)
    sl = slice(idx[0], idx[-1] + 1)
    return Datacube(
        flux=cube.flux[sl],
        spectral_axis=axis[sl],
        spectral_kind=cube.spectral_kind,
        pixel_scale=cube.pixel_scale,
        redshift=cube.redshift,
        flux_unit=cube.flux_unit,
        variance=cube.variance[sl] if cube.variance is not None else None,
        mask=cube.mask[sl] if cube.mask is not None else None,
        meta=dict(cube.meta),
    )

    kino.analysis.fisher ¶

    Fast shear-precision forecast: a Fisher (Cramer-Rao) bound through the same forward model as the inference likelihoods.

    For a galaxy, instrument and exposure this estimates how well (g1, g2) can be measured without running a sampler (a few forward renders instead of thousands), which is what makes it usable for ranking hundreds of targets over a semester.

    Two estimators (shear_fisher(..., estimator=...)):

    "cube" (default) The bound of :class:kino.inference.likelihood.IFUCubeLikelihood: the model cube mu (expected electrons, sky + dark baseline included) against the data voxel by voxel with var = mu + read^2, F_ij = sum dmu_i dmu_j (1/var + 1/(2 var^2)) + F_prior. Unbiased, and it uses the flux image as well as the kinematics.

    "moments" The bound of :class:kino.inference.likelihood.IFUMomentLikelihood: derivatives of the flux-weighted Vlos and dispersion maps weighted by their delta-method errors. Kept for comparison; the moment estimator is noise-biased (below).

    Two numbers come out of either: sigma_fixed (only the shear free -- optimistic) and sigma_marginal (nuisance parameters free under priors -- what a real fit faces).

    What was validated (see scripts/validate_fisher_mle.py, scripts/validate_fisher_forecast.py and notebooks/data/fisher_validation.json), against maximum-likelihood fits on many noise realisations:

    • cube: maximum-likelihood shear unbiased to ~1e-4 and scatter / Fisher = 0.9-1.3, at every per-spaxel S/N tested (about 5 to 100), including with inclination, PA and v_flat free.
    • moments: scatter / sigma_fixed = 1.01 at a median valid-spaxel S/N of ~23, 1.28 at ~13, 2.2-3.6 at ~7; and a noise bias of ~1 sigma per galaxy that grows toward low S/N and that a stack does not average away. An analytic second-order bias correction fails below S/N ~ 12. median_spaxel_snr and :data:LOW_SNR_WARNING flag the regime.

    What the bound is not. It is a statistical Cramer-Rao bound for a perfect model: the data are generated by the same DiscGalaxy family the fit assumes, so the data pin the nuisance parameters (inclination to ~0.1 deg, ...) far below any prior, and the marginal shear precision comes out 1e-3-1e-2 per galaxy for bright targets. Real galaxies have non-circular motions, thickness, warps, an uncertain PSF; that sets a floor the bound ignores. Add one with forecast_ifu(intrinsic_sigma=...) and psf_error=. The bound is also a local, Gaussian approximation: it says nothing about multi-modal posteriors.

    :func:shear_fisher_multi generalises the cube estimator to several independent exposures of the same galaxy -- e.g. a slit visited at a few position angles/offsets to cover more of its velocity field, rather than one exposure at a single, possibly misaligned, position angle. Each exposure's own voxel-level Fisher information is summed before the shared nuisance priors are applied once. It does not (yet) support psf_error or the moment estimator.

    def shear_snr(self, g: tuple[float, float], *, marginal: bool = True) -> float:
    """``|g| / sigma`` along the direction of ``g`` (or the mean sigma if ``g = 0``)."""
    cov = self.covariance[:2, :2] if marginal else np.diag(np.square(self.sigma_fixed))
    gv = np.asarray(g, dtype=float)
    norm = float(np.linalg.norm(gv))
    if norm == 0.0:
        return 0.0
    unit = gv / norm
    return norm / float(np.sqrt(unit @ cov @ unit))

    *,

    g: tuple[float, float] = (0.0, 0.0),
    snr_threshold: float = 3.0,
    priors: dict[str, float | None] | None = None,
    steps: dict[str, float] | None = None,
    free: list[str] | None = None,
) -> FisherResult:
    """Cube-estimator Fisher forecast for **several independent exposures of the same
    galaxy** -- e.g. a slit visited at a few different position angles or offsets to
    cover more of its velocity field, or repeat visits on different nights.

    Each exposure's own voxel-level Fisher information (statistically independent noise
    realisations of the *same* underlying galaxy/shear) is summed before the shared
    nuisance priors are applied **once**, not once per exposure -- the priors describe
    the one true galaxy, not each observation of it:

    ``F_total = sum_k F_k``, then ``cov = (F_total + prior precision)^-1``.

    Parameters
    ----------
    instruments : list of IFU or SlitSpectrograph
        Two or more instruments, each with its own exposure time/sky/PSF/slit geometry
        already set (e.g. different ``slit_pa`` values -- split a fixed total time
        budget across them yourself, such as ``exposure_time`` scaled by ``1/len(...)``
        on each). All observe the *same* galaxy and shear point; parameter names/steps
        are shared, so a duck-typing mismatch (mixing an IFU and a SlitSpectrograph) is
        allowed as long as both expose the same ``.observe()``/``.baseline_electrons()``
        API used by :func:`shear_fisher`.
    line_flux, line_wavelength, g, snr_threshold, priors, steps, free
        As in :func:`shear_fisher`.

    Returns
    -------
    FisherResult
        ``n_spaxels`` is summed across exposures; ``median_spaxel_snr``/
        ``peak_spaxel_snr`` are computed over the pooled per-spaxel S/N of every
        exposure's used spaxels; ``line_snr`` combines each exposure's own whole-field
        S/N in quadrature (exact only if every exposure captures the same total flux,
        which need not hold for different slit position angles on an inclined disc --
        treat it as a descriptive summary, not something the covariance depends on).

    Notes
    -----
    Only the cube estimator generalises simply to several exposures (the moment
    estimator's per-exposure noise bias does not combine this way) and ``psf_error``/
    ``psf_bias`` is not supported yet (always NaN) -- extending the joint
    mis-specified-PSF fit to several exposures is future work, not implemented here.
    """
    if not instruments:
        raise ValueError("instruments must be non-empty")
    priors = {**DEFAULT_CUBE_PRIORS, **(priors or {})}
    steps = {**_CUBE_STEPS, **(steps or {})}
    base, names, theta0 = _cube_names_and_theta0(galaxy, g, priors, free)
    n = len(names)

    fisher_total = np.zeros((n, n))
    n_used = 0
    snr_pool: list[np.ndarray] = []
    line_snr_sq = 0.0
    for instrument in instruments:
        fid, cols, baseline, _ = _cube_jacobian(
            instrument, base, names, theta0, line_flux, line_wavelength, steps
        )
        var = np.maximum(fid.variance, _VARIANCE_FLOOR)
        signal = fid.flux - baseline
        spaxel_snr = signal.sum(axis=0) / np.sqrt(var.sum(axis=0))
        used = spaxel_snr >= snr_threshold
        if used.any():
            n_used += int(used.sum())
            snr_pool.append(spaxel_snr[used])
        fisher_total += _cube_information(cols, var)
        line_snr_sq += float(signal.sum() ** 2 / var.sum())

    if n_used == 0:
        raise ValueError("no spaxel passes the flux S/N threshold -- the target is undetectable")
    pooled_snr = np.concatenate(snr_pool)

    cov, fixed, _ = _invert_with_priors(fisher_total, names, theta0, priors)
    return FisherResult(
        sigma_fixed=(float(np.sqrt(fixed[0, 0])), float(np.sqrt(fixed[1, 1]))),
        sigma_marginal=(float(np.sqrt(cov[0, 0])), float(np.sqrt(cov[1, 1]))),
        covariance=cov,
        param_names=names,
        n_spaxels=n_used,
        line_snr=float(np.sqrt(line_snr_sq)),
        peak_spaxel_snr=float(np.max(pooled_snr)),
        median_spaxel_snr=float(np.median(pooled_snr)),
        meta={"estimator": "cube", "n_exposures": len(instruments)},
    )

    kino.analysis.forecast ¶

    Target-level forecast: shear S/N as a function of seeing, for a given night.

    forecast_ifu and forecast_slit are the entry points that tie the observing-conditions layer (:mod:kino.observing) to an instrument backend (an :class:~kino.instruments.ifu.IFU or a :class:~kino.instruments.slit.SlitSpectrograph) and the fast Fisher estimate (:mod:kino.analysis.fisher). Given

    • an instrument (its exposure, throughput, detector, PSF machinery),
    • a target: sky position, a :class:~kino.galaxy.DiscGalaxy (geometry and redshift) and the line flux measured from its spectrum (:func:kino.io.spectrum.measure_line_flux),
    • a night, and the shear you expect,

    each returns, at times through the night and for each seeing you ask about, the airmass, the sky, the effective seeing, the line S/N and the shear precision and S/N -- all computed with the sky, moon, extinction and seeing appropriate to that moment. Both are thin wrappers around _forecast (below), which does the actual work and is agnostic to which instrument type it was given.

    What happens at each (time, seeing) point:

    1. airmass, moon phase/altitude/separation and ecliptic position from the target's track (kino.observing);
    2. a sky spectrum from the offline SkyCalc library, converted to detected electrons per channel on the instrument's own collecting area/throughput (per spaxel for an IFU, per along-slit pixel for a slit);
    3. the line flux dimmed by CTIO atmospheric extinction (include_extinction);
    4. the quoted (500 nm, zenith) seeing converted to the observed wavelength and airmass (Kolmogorov scaling), set as the instrument's seeing PSF;
    5. :func:~kino.analysis.fisher.shear_fisher on the noiseless expected cube (the "cube" estimator works for either instrument; slit losses are inside the model exactly as they are inside a real slit observation).

    Assumptions you must be comfortable with (each is also in CLAUDE.md):

    • the instrument's throughput curve is end to end excluding the atmosphere; extinction is applied separately. If your throughput already includes the atmosphere (a standard-star efficiency often does), pass include_extinction=False;
    • the sky model is Paranal's, not the site's; airglow is a whole-night mean;
    • line_flux is the whole galaxy's intrinsic flux (aperture-corrected, magnification neglected -- a few per cent for weak lensing);
    • only the emission line constrains the shear -- no continuum;
    • the spectral window is cropped to window_kms around the line, keeping the instrument's native dispersion, purely so the many renders a forecast needs stay fast -- not a physical limitation (see narrowed_ifu/narrowed_slit);
    • the primary, default forecast (intrinsic_sigma=None) is the idealised statistical bound: real-galaxy floor calibration (non-circular motions, disc thickness, ...) is out of scope for now (2026-09-27 direction) and left as an optional sensitivity comparison, not a required input.

    kino.analysis.injection ¶

    Injection-recovery: apply a known shear, re-observe, and check recovery.

    Implements README.md Phase 6's injection-recovery, shape-noise, (m, c) bias, and null-test bullets for the IFU case. Each trial takes an intrinsic :class:~kino.galaxy.DiscGalaxy, applies a known reduced shear (g1, g2) via :func:kino.lensing.apply_shear, observes it through an :class:~kino.instruments.ifu.IFU (one noisy realisation), runs :func:kino.inference.fit_shear, and records recovered vs. input shear. :func:summarize_trials then reduces an ensemble of trials to the calibration numbers a shear measurement needs:

    • multiplicative/additive bias (m, c) from g_rec = (1 + m) g_true + c (needs trials at two or more distinct true shears),
    • residual shape noise sigma_eps, the per-galaxy scatter of g_rec - g_true (the shear-equivalent noise of one galaxy),
    • the mean bias at a single shear (with g_true = 0 this is the null test),
    • the pull (g_rec - g_true) / sigma_post, whose standard deviation should be ~1 if the posteriors' widths are honest.
    Design notes
    • The input is a DiscGalaxy, not a Datacube as the original stub sketched: a cube is already an observation and cannot be re-sheared, so injection needs the underlying model. (Known-truth cube recovery, e.g. from kino.io.kinms, is a different check -- fitting an unlensed model to a cube -- and is not what this module does.)
    • Recovered shear is the posterior median. With a finite-width uniform prior and weakly-constraining data the median is pulled toward the prior centre, so a measured m < 0 (shrinkage) is expected at low S/N and is a property of the estimator + prior, not necessarily a forward-model bug -- compare m across S/N before interpreting it.
    • Every trial gets its own child seed (numpy.random.SeedSequence.spawn) for both the noise realisation and the sampler, so a run is fully reproducible from one seed and trials are independent of ensemble order/size. Trials are embarrassingly parallel; callers wanting to parallelise can call :func:recover_shear themselves with their own pool.

    """Inject ``(g1, g2)`` into ``galaxy``, observe it with ``ifu``, and infer it back.

    Parameters
    ----------
    galaxy : DiscGalaxy
        The intrinsic (unlensed) truth.
    g1, g2 : float
        Injected reduced shear.
    ifu : IFU
        Instrument used both to make the data and (inside the likelihood) to
        forward-model every trial point -- see
        :class:`~kino.inference.likelihood.IFUMomentLikelihood`.
    line_wavelength : Quantity (wavelength)
        Rest-frame line wavelength.
    line_flux : Quantity (energy / time / area)
        Intrinsic (unlensed) line flux, assumed known (fixed in the fit).
    free_priors : dict or callable, optional
        Priors (in :func:`~kino.inference.priors.build_prior`'s format) for
        free parameters *other than* the shear -- e.g. ``{"inc": (30, 80)}``.
        A callable ``f(galaxy) -> dict`` lets priors depend on the galaxy
        (e.g. a Tully-Fisher prior centred on its flux). Every
        ``DiscGalaxy`` field not named here is held fixed at the galaxy's
        true value. Default: shear-only fit.
    shear_prior_range : float, optional
        Uniform prior half-width on both ``g1`` and ``g2``.
    rng : numpy.random.Generator or None, optional
        Drives the data's noise realisation; ``None`` gives noiseless data.
    sampler_seed : int or None, optional
        Seed for ``nautilus``.
    likelihood : {"cube", "moments"}, optional
        ``"cube"`` (default): :class:`~kino.inference.likelihood.IFUCubeLikelihood`, unbiased.
        ``"moments"``: :class:`~kino.inference.likelihood.IFUMomentLikelihood`, whose moment
        estimator has a noise bias at low S/N (see ``CLAUDE.md``).
    snr_threshold, n_live, n_eff : optional
        ``snr_threshold`` is used by the moment likelihood only; the rest are forwarded
        to :func:`~kino.inference.sampler.fit_shear`.
    keep_posterior : bool, optional
        Store the full :class:`~kino.inference.sampler.Posterior` on the trial.

    Returns
    -------
    InjectionTrial
    """

        apply_shear(galaxy, g1, g2), line_flux=line_flux, line_wavelength=line_wavelength, rng=rng

    extra_priors = _resolve_free_priors(free_priors, galaxy)
    bad = [k for k in extra_priors if k not in GALAXY_FIELDS]
    if bad:
        raise ValueError(f"free_priors may only name DiscGalaxy fields, got {bad}")
    fixed = {f: getattr(galaxy, f) for f in GALAXY_FIELDS if f not in extra_priors}

    if likelihood == "cube":
        lik = IFUCubeLikelihood(
            ifu=ifu,
            line_wavelength=line_wavelength,
            data=data,
            fixed_galaxy_params=fixed,
            fixed_line_flux=line_flux,
        )
    elif likelihood == "moments":
        lik = IFUMomentLikelihood(
            ifu=ifu,
            line_wavelength=line_wavelength,
            data=data,
            fixed_galaxy_params=fixed,
            fixed_line_flux=line_flux,
            snr_threshold=snr_threshold,
        )
    else:
        raise ValueError(f"likelihood must be 'cube' or 'moments', got {likelihood!r}")

            "g2": (-shear_prior_range, shear_prior_range),

    )

        g_rec=g_rec,
        g_err=g_err,
        galaxy=galaxy,
        posterior=posterior if keep_posterior else None,
    )
