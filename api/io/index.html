    kino.io¶

    The Datacube container plus readers/generators (KinMS, MaNGIA), real-spectrum utilities, and FITS/HDF5 I/O for KINO's own mock data products (kino.io.fitsio/kino.io.hdf5, README.md Phase 4).

    Read/write of mock data and results.

    The common Datacube container (kino.io.base) plus readers/generators that convert each data source into it (kino.io.kinms, kino.io.manga) and the test-data downloader (kino.io.download) -- see TEST_DATA.md. kino.io.fitsio/kino.io.hdf5 are FITS/HDF5 I/O for KINO's own mocks and run provenance (README.md Phase 4), with the same four function names on each backend.

    def write_slit_observation(obs: SlitObservation, path: str | Path) -> None:
    """Write ``obs`` to ``path`` as a FITS file (see module docstring)."""
    primary = fits.PrimaryHDU()
    hdus: list[fits.hdu.base._BaseHDU] = [primary, *_cube_to_hdus(obs.spectrum, "SPECTRUM")]
    image_hdu = fits.ImageHDU(data=np.asarray(obs.image, dtype=np.float64), name="IMAGE")
    ih = image_hdu.header
    ih["IMGSCALE"] = obs.image_pixel_scale.to_value(u.arcsec)
    ih["SLITW"] = obs.slit_width.to_value(u.arcsec)
    ih["SLITLEN"] = obs.slit_length.to_value(u.arcsec)
    ih["SLITPA"] = obs.slit_pa
    ih["SLITOFFX"] = obs.slit_offset[0].to_value(u.arcsec)
    ih["SLITOFFY"] = obs.slit_offset[1].to_value(u.arcsec)
    hdus.append(image_hdu)
    fits.HDUList(hdus).writeto(path, overwrite=True)

    kino.io.spectrum ¶

    1-D galaxy spectra from tables, and emission-line flux measurement.

    A forecast needs the galaxy's line flux and redshift. Spectra come from public archives (Legacy Survey/DESI/SDSS, southern surveys, ...) as ASCII or FITS tables in differing conventions, so the reader is deliberately strict about units: a flux column with no unit in the file or in the call raises instead of guessing (SDSS/DESI's 1e-17 erg/s/cm2/A convention is a classic silent factor-of-1e17 error).

    Wavelengths are treated as vacuum; SDSS and DESI spectra are vacuum, many other sources quote air wavelengths -- convert first (the ~2 A shift at H-alpha is irrelevant to the line flux but matters for sky-line overlap).

    The measured flux is what the spectrograph aperture (a fibre, a slit) collected, not the galaxy's total line flux. Pass aperture_correction when you know the fraction of the galaxy inside that aperture.

    def read_spectrum(
    path: str | Path,
    *,
    wavelength_col: str | None = None,
    flux_col: str | None = None,
    error_col: str | None = None,
    wavelength_unit: u.UnitBase | str | None = None,
    flux_unit: u.UnitBase | str | None = None,
    hdu: int | str = 1,
    format: str | None = None,  # noqa: A002 -- astropy's own keyword
) -> Spectrum1D:
    """Read a spectrum from an ASCII or FITS binary table.

    Columns are found by common names (wavelength: ``wavelength/wave/lam``;
    flux: ``flux``; error: ``error/err/flux_err`` or ``ivar``) unless named.
    ``loglam`` (SDSS, log10 of the wavelength in Angstrom) is converted.
    ``wavelength_unit`` and ``flux_unit`` are used when the file carries no
    unit; **a missing flux unit raises** rather than defaulting.

    ``.fits``/``.fit`` files are read with ``hdu``; anything else through
    ``astropy.io.ascii`` (whitespace, CSV, ...; ``#`` comment lines are fine).
    A header-less two- or three-column text file is read as
    ``wavelength flux [error]``.
    """
    path = Path(path)
    if path.suffix.lower() in (".fits", ".fit", ".fits.gz"):
        table = Table.read(path, hdu=hdu)
    else:
        table = Table.read(path, format=format or "ascii")
        if all(c.startswith("col") for c in table.colnames):
            table.rename_columns(
                table.colnames, ["wavelength", "flux", "error"][: len(table.colnames)]
            )
    lam_name = _find(table, _WAVE_NAMES, wavelength_col)
    lam_is_log = False
    if lam_name is None and "loglam" in {c.lower() for c in table.colnames}:
        lam_name = next(c for c in table.colnames if c.lower() == "loglam")
        lam_is_log = True
    flux_name = _find(table, _FLUX_NAMES, flux_col)
    if lam_name is None or flux_name is None:
        raise ValueError(f"cannot identify wavelength/flux columns in {table.colnames}")

    lam = np.asarray(table[lam_name], dtype=float)
    if lam_is_log:
        lam, w_unit = 10.0**lam, u.AA
    else:
        w_unit = table[lam_name].unit or (
            None if wavelength_unit is None else u.Unit(wavelength_unit)
        )
        if w_unit is None:
            raise ValueError("wavelength has no unit: pass wavelength_unit (e.g. 'Angstrom')")
    f_unit = table[flux_name].unit or (None if flux_unit is None else u.Unit(flux_unit))
    if f_unit is None:
        raise ValueError(
            "flux has no unit in the file: pass flux_unit explicitly "
            "(e.g. 1e-17 * u.erg / u.s / u.cm**2 / u.AA for SDSS/DESI)"
        )
    if flux_unit is not None and table[flux_name].unit is not None and f_unit != u.Unit(flux_unit):
        raise ValueError(f"file says {f_unit}, call says {flux_unit}: refusing to guess")
    flux = np.asarray(table[flux_name], dtype=float) * f_unit

    err = None
    err_name = _find(table, _ERR_NAMES, error_col)
    if err_name is not None:
        e_unit = table[err_name].unit or f_unit
        err = np.asarray(table[err_name], dtype=float) * e_unit
    elif "ivar" in {c.lower() for c in table.colnames}:
        ivar = np.asarray(
            table[next(c for c in table.colnames if c.lower() == "ivar")], dtype=float
        )
        with np.errstate(divide="ignore"):
            err = np.where(ivar > 0, 1.0 / np.sqrt(ivar), np.inf) * f_unit

    good = np.isfinite(lam) & np.isfinite(flux.value)
    order = np.argsort(lam[good])
    lam_q = (lam[good][order] * w_unit).to(u.AA)
    return Spectrum1D(
        wavelength=lam_q,
        flux=flux[good][order],
        error=None if err is None else err[good][order],
        meta={"source": str(path)},
    )

    def measure_line_flux(
    spectrum: Spectrum1D,
    redshift: float,
    line: str | u.Quantity = "Halpha",
    *,
    half_width: u.Quantity = 1500.0 * u.km / u.s,
    aperture_correction: float = 1.0,
) -> LineMeasurement:
    """Integrated flux of one emission line: Gaussian plus linear continuum.

    Fits ``amp * exp(-(lam - mu)^2 / 2 sigma^2) + c0 + c1 (lam - mu)`` in
    ``+/- half_width`` (velocity) around the expected observed wavelength and
    integrates the Gaussian: ``F = amp * sigma * sqrt(2 pi)``. Uses the
    spectrum's errors if present, otherwise the scatter of the residuals.

    ``aperture_correction`` (>= 1) multiplies the flux to estimate the whole
    galaxy's line flux from what the aperture collected; 1 leaves it as
    measured. The result is *not* corrected for Milky Way or internal dust.
    """
    rest = LINES[line] * u.AA if isinstance(line, str) else u.Quantity(line).to(u.AA)
    if aperture_correction < 1.0:
        raise ValueError("aperture_correction must be >= 1")
    expected = rest * (1.0 + redshift)
    dv = (half_width / (C_KMS * u.km / u.s)).decompose().value
    lo, hi = expected * (1 - dv), expected * (1 + dv)
    lam_all = spectrum.wavelength.to_value(u.AA)
    sel = (lam_all >= lo.value) & (lam_all <= hi.value)
    if sel.sum() < 8:
        raise ValueError(
            f"only {int(sel.sum())} spectrum pixels within +/-{half_width} of the expected "
            f"line at {expected:.1f} -- is the line inside the spectrum's coverage?"
        )
    lam = lam_all[sel]
    f_unit = spectrum.flux.unit
    y = spectrum.flux.value[sel]
    sigma_y = None
    if spectrum.error is not None:
        e = spectrum.error.to_value(f_unit)[sel]
        if np.all(np.isfinite(e)) and np.all(e > 0):
            sigma_y = e

    pix = float(np.median(np.diff(lam)))
    sig0 = max(1.0 * pix, expected.value * 100.0 / C_KMS)
    c0 = float(np.median(y))
    amp0 = float(np.max(y) - c0)
    p0 = [amp0, expected.value, sig0, c0, 0.0]
    try:
        popt, pcov = curve_fit(
            _gauss_line, lam, y, p0=p0, sigma=sigma_y, absolute_sigma=sigma_y is not None,
            maxfev=20000,
        )  # fmt: skip
    except RuntimeError as exc:
        raise ValueError(f"line fit did not converge: {exc}") from exc
    amp, mu, sig = popt[0], popt[1], abs(popt[2])
    # with sigma=None curve_fit already scales pcov by the residual variance
    # F = amp * sigma * sqrt(2 pi); propagate including the amp-sigma covariance
    grad = np.array([sig, amp]) * np.sqrt(2 * np.pi)
    cov2 = np.array([[pcov[0, 0], pcov[0, 2]], [pcov[2, 0], pcov[2, 2]]])
    var_f = float(grad @ cov2 @ grad)
    flux = amp * sig * np.sqrt(2 * np.pi) * f_unit * u.AA
    flux_err = np.sqrt(max(var_f, 0.0)) * f_unit * u.AA
    unit = u.erg / u.s / u.cm**2
    f = (flux * aperture_correction).to(unit)
    fe = (flux_err * aperture_correction).to(unit)
    return LineMeasurement(
        flux=f,
        flux_err=fe,
        center=mu * u.AA,

    kino.io.kinms ¶

    Generate known-truth datacubes with KinMS, for injection-recovery tests.

    Thin wrapper around the kinms package (Davis et al. 2013, MNRAS 429, 534; https://github.com/TimothyADavis/KinMSpy) that maps KINO's documented conventions (CLAUDE.md) onto KinMS's cube-generation API and returns a :class:kino.io.base.Datacube. This is Tier A of the validation ladder in TEST_DATA.md: cubes with an exactly known input, used for unit and injection-recovery tests before touching any real or simulated survey data.

    Position-angle convention

    KinMS's posAng places its disc's redshifted lobe along +y at posAng=0 and rotates it towards -x as posAng increases (verified empirically — KinMS ships no explicit statement of rotation sense). KINO's pa is degrees East of North in the (x=East, y=North) plane, i.e. it rotates from +y towards +x as pa increases. The two rotate in opposite senses, so this wrapper passes posAng=-pa to KinMS to align the kinematic major-axis line with KINO's convention.

    The major-axis line is pinned by tests/io/test_kinms.py::test_position_angle_convention. Which end of that line is redshifted is confirmed consistent with kino.galaxy's independently-derived Vlos convention (Phase 1) by tests/galaxy/test_disc.py::test_pa_zero_redshifted_side_matches_kinms_convention and, at the level of a full map rather than two points, by tests/galaxy/test_kinms_consistency.py (Vlos-map correlation > 0.99 against an independently generated KinMS cube).

    Parameters
    ----------
    field_of_view : Quantity (angle)
        Square spatial extent of the output cube.
    pixel_scale : Quantity (angle)
        Spaxel size.
    velocity_range : Quantity (speed)
        Extent of the spectral (velocity) axis.
    channel_width : Quantity (speed)
        Channel width of the spectral axis.
    beam_size : Quantity (angle), scalar

        supported by this thin wrapper; pass an elliptical/angled beam
        directly to ``kinms.KinMS`` if needed.
    inc : float
        Inclination, degrees; 0 = face-on, 90 = edge-on (``CLAUDE.md``).
    pa : float
        Position angle, degrees East of North (``CLAUDE.md``); see the
        module docstring for how this maps onto KinMS's ``posAng``.
    sb_rad : Quantity (angle)
        Radius vector for the surface-brightness profile.
    sb_prof : array-like
        Surface-brightness profile (arbitrary scale) vs. ``sb_rad``.
    vel_rad : Quantity (angle)
        Radius vector for the circular-velocity profile.
    vel_prof : Quantity (speed)
        Circular velocity profile vs. ``vel_rad``.
    gas_sigma : Quantity (speed), optional
        Intrinsic (constant) velocity dispersion.
    v_sys : Quantity (speed), optional
        Systemic velocity, added to the spectral axis's zero point.
    redshift : float, optional
        Cosmological redshift recorded on the returned ``Datacube`` as
        metadata only — KinMS itself models in the local/rest frame.
    n_clouds : int, optional
        Number of point-cloud samples (KinMS ``nSamps``); higher reduces
        Monte Carlo noise in the cube at the cost of runtime.
    rng : numpy.random.Generator or None, optional
        Seeded RNG (``CLAUDE.md``: no global RNG use) used to draw a scalar
        seed for KinMS's ``seed`` argument (KinMS internally scales a single
        int into its 4 per-stream seeds -- do not pass a 4-element seed here,
        it silently overflows to an invalid seed for each stream).
        ``None`` uses KinMS's own default seed.

    Returns
    -------
    Datacube
        ``flux`` in arbitrary units (KinMS's default ``Jy/beam``-scaled
        output; no absolute flux calibration is applied). ``spectral_axis``
        is velocity (km/s), linearly spaced and centred on ``v_sys``.
    """
    seed = None
    if rng is not None:
        seed = int(rng.integers(1, 1_000_000))

    model = KinMS(
        xs=field_of_view.to_value(u.arcsec),
        ys=field_of_view.to_value(u.arcsec),
        vs=velocity_range.to_value(u.km / u.s),
        cellSize=pixel_scale.to_value(u.arcsec),
        dv=channel_width.to_value(u.km / u.s),
        beamSize=beam_size.to_value(u.arcsec),
        nSamps=n_clouds,
        seed=seed,
        fixSeed=seed is not None,
        verbose=False,
    )
    cube_xyv = model.model_cube(
        inc=inc,
        posAng=-pa,  # KinMS posAng rotates opposite to KINO's pa -- see module docstring
        gasSigma=gas_sigma.to_value(u.km / u.s),

        sbRad=sb_rad.to_value(u.arcsec),
        velRad=vel_rad.to_value(u.arcsec),
        velProf=vel_prof.to_value(u.km / u.s),
        vSys=v_sys.to_value(u.km / u.s),
    )
    # KinMS returns (n_x, n_y, n_v); Datacube wants (n_spec, n_y, n_x).
    flux = np.transpose(cube_xyv, (2, 1, 0))

    n_chan = flux.shape[0]
    dv = channel_width.to_value(u.km / u.s)
    spectral_axis = ((np.arange(n_chan) - (n_chan - 1) / 2) * dv + v_sys.to_value(u.km / u.s)) * (
        u.km / u.s
    )

        flux=flux,
        spectral_axis=spectral_axis,
        spectral_kind="velocity",
        pixel_scale=pixel_scale.to(u.arcsec),
        redshift=redshift,
        flux_unit=u.dimensionless_unscaled,

            "citation": "Davis et al. 2013, MNRAS 429, 534",
            "generator_params": {
                "inc_deg": inc,
                "pa_deg": pa,
                "v_sys_kms": v_sys.to_value(u.km / u.s),
                "n_clouds": n_clouds,
            },
        },
    )

    • "rss" (:func:read_mangia_rss -> :class:MangiaRSS): a genuine flux-calibrated spectrum per fibre (shape (n_wave, n_fiber), real units, a real wavelength grid) plus each fibre's real sky position -- but not pre-gridded into a cube. :func:regrid_rss_to_cube turns this into a real :class:~kino.io.base.Datacube via a simplified Gaussian-kernel ("drizzle"-like) regridding -- see its own docstring for exactly what it does and doesn't match about the official MaNGA DRP cubing algorithm (Law et al. 2016) or MaNGIA's own generator code (reginasar/TNG_MaNGA_mocks, GPL-3.0), which this does not call or port, only take inspiration from.

    maps (maps/TNG50-{snap}-{subhalo}-{view}-{n_fibers}.cube_maps.fits): INTRINSIC_ASSIGNED (primary): shape (41, ny, nx), one 2-D map per TYPE<i> header key (e.g. TYPE2 = VEL) -- simulation ground truth, not a pyPipe3D fit. Header: CD1_1/CD2_2 (pixel scale, degrees), REDSHIFT, PSF (fibre PSF FWHM, arcsec), FOV (arcsec).     rss (RSS/TNG50-{snap}-{subhalo}-{view}-{n_fibers}.cube_RSS.fits): Primary: flux, shape (n_wave=6603, n_fiber), UNITS header says 1E-16 erg/s/cm^2 (assumed per-Angstrom, i.e. the implicit unit is 1e-16 erg/s/cm^2/Angstrom -- the header does not spell out the /Angstrom, so this is still an assumption, just a far better grounded one than the previous version's).     def regrid_fibers_to_grid(
    x_fiber: u.Quantity,
    y_fiber: u.Quantity,
    values: np.ndarray,
    *,
    pixel_scale: u.Quantity,
    nx: int,
    ny: int,
    kernel_fwhm: u.Quantity,
    x0: u.Quantity = 0.0 * u.arcsec,
    y0: u.Quantity = 0.0 * u.arcsec,
) -> np.ndarray:
    """Regrid per-fibre ``values`` (last axis = fibre) onto a regular
    ``(..., ny, nx)`` grid by a Gaussian-kernel weighted average --
    ``out[..., y, x] = sum_f w_f(x,y) * values[..., f] / sum_f w_f(x,y)``,
    ``w_f(x,y) = exp(-0.5 * dist((x,y),(x_f,y_f))^2 / sigma^2)``,
    ``sigma = kernel_fwhm / 2.3548``.

    This is a simplified "drizzle"-like regridding, **not** a reproduction
    of the real MaNGA DRP cubing algorithm (Law et al. 2016) or MaNGIA's
    own generator (``reginasar/TNG_MaNGA_mocks``): it ignores each fibre's
    finite aperture (treating it as a point sample) and does not conserve
    flux through any aperture-overlap accounting. **It also has no fibre-
    bundle footprint mask** -- the Gaussian kernel has no hard cutoff, so
    output pixels outside the real fibre coverage still get a (smoothly
    extrapolated, increasingly unreliable) value rather than being flagged
    invalid; mask the output to the fibre bundle's own footprint yourself
    if you need one (e.g. a matching ``MangiaMaps`` map's own ``!= 0``
    footprint, as ``notebooks/06_mangia_datacube_demo.ipynb`` and
    ``scripts/validate_mangia_regrid.py`` do). Validated empirically (not
    by construction) against MaNGIA's own pre-gridded ``maps`` product for
    the same galaxies, within that footprint -- see ``notebooks/`` and
    ``tests/io/test_manga.py`` for the comparison.

    Parameters
    ----------
    x_fiber, y_fiber : Quantity (angle), shape (n_fiber,)
    values : numpy.ndarray, shape (..., n_fiber)
    pixel_scale : Quantity (angle)
    nx, ny : int
    kernel_fwhm : Quantity (angle)
    x0, y0 : Quantity (angle), optional
        Grid centre, same convention as ``x_fiber``/``y_fiber``.

    Returns
    -------
    numpy.ndarray, shape (..., ny, nx)
    """
    n_fiber = x_fiber.shape[0]
    if values.shape[-1] != n_fiber:
        raise ValueError(
            f"values' last axis ({values.shape[-1]}) must match the number of fibres ({n_fiber})"
        )
    px = pixel_scale.to_value(u.arcsec)
    xf = x_fiber.to_value(u.arcsec)
    yf = y_fiber.to_value(u.arcsec)
    x_axis = (np.arange(nx) - (nx - 1) / 2.0) * px + x0.to_value(u.arcsec)
    y_axis = (np.arange(ny) - (ny - 1) / 2.0) * px + y0.to_value(u.arcsec)
    xx, yy = np.meshgrid(x_axis, y_axis)  # (ny, nx)

    sigma = kernel_fwhm.to_value(u.arcsec) / (2.0 * np.sqrt(2.0 * np.log(2.0)))
    dx = xx[..., np.newaxis] - xf[np.newaxis, np.newaxis, :]
    dy = yy[..., np.newaxis] - yf[np.newaxis, np.newaxis, :]
    w = np.exp(-0.5 * (dx**2 + dy**2) / sigma**2)  # (ny, nx, n_fiber)
    w_sum = w.sum(axis=-1)
    w_sum_safe = np.where(w_sum > 0, w_sum, 1.0)

    out = np.einsum("...f,yxf->...yx", values, w) / w_sum_safe
    return np.where(w_sum > 0, out, 0.0)

    in **one streaming pass**, writing each to its own destination in
    ``members`` (``{member name: dest path}``).

    Stops reading the remote stream as soon as every requested member has
    been found, so asking for several members scattered through a large
    archive costs one scan, not one per member (the naive per-member
    approach :func:`stream_extract_tar_member` uses, fine for one member

    unless ``overwrite=True``, and are not searched for in the stream.

    Returns
    -------
    dict[str, Path]
        ``{member name: dest path}`` for every member actually extracted
        (or already cached) this call -- members not found in the archive
        are simply absent, not an error (callers can diff against
        ``members`` to see what's missing).
    """
    pending = {
        name: Path(dest) for name, dest in members.items() if overwrite or not Path(dest).exists()
    }
    found: dict[str, Path] = {
        name: Path(dest) for name, dest in members.items() if name not in pending
    }
    if not pending:
        return found

    req = urllib.request.Request(url, headers=headers or {})
    with urllib.request.urlopen(req) as resp, tarfile.open(fileobj=resp, mode="r|gz") as tf:
        for entry in tf:
            if entry.name not in pending:
                continue
            extracted = tf.extractfile(entry)
            if extracted is None:
                continue  # not a regular file (e.g. a directory entry)
            dest = pending.pop(entry.name)
            dest.parent.mkdir(parents=True, exist_ok=True)
            with open(dest, "wb") as out:
                shutil.copyfileobj(extracted, out)
            found[entry.name] = dest
            if not pending:
                break
    return found
