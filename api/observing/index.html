    kino.observing¶

    Observing conditions — site, night, airmass, moon, seeing, extinction, and the sky spectrum model — sitting between a target and an instrument backend. Knows nothing about instruments.

    Observing conditions: where the target is, when it is observable, and what the atmosphere and moon are doing.

    This layer sits between a target and an instrument backend: it turns "RA/Dec on this night" into airmass, moon state and effective seeing, which later feed the instrument's throughput, sky background and PSF. It knows nothing about instruments. See README.md (observing-conditions bullets).

    @classmethod
def load(cls, path: str | Path = DEFAULT_LIBRARY_PATH) -> SkyLibrary:
    """Load the library built by ``scripts/build_skycalc_library.py``."""
    path = Path(path)
    try:
        moon = dict(np.load(path / "moon.npz"))
        airglow = dict(np.load(path / "airglow.npz"))
        zodi = dict(np.load(path / "zodi.npz"))
    except FileNotFoundError as exc:
        raise FileNotFoundError(
            f"no sky library in {path}/ -- build it once (needs network and "
            "`pip install skycalc-cli`): python scripts/build_skycalc_library.py"
        ) from exc
    meta = {}
    if (path / "meta.json").exists():
        import json

        meta = json.loads((path / "meta.json").read_text())
    return cls(moon, airglow, zodi, meta)

    def spectrum(self, conditions: SkyConditions, wavelength_edges: u.Quantity) -> SkySpectrum:
    """Sky components averaged over the channels bounded by ``wavelength_edges``.

    ``wavelength_edges`` are *vacuum* wavelengths, shape ``(n + 1,)``.
    """
    if conditions.airmass < 1.0:
        raise ValueError("airmass must be >= 1")
    edges_nm = u.Quantity(wavelength_edges).to_value(u.nm)
    if edges_nm[0] < _MIN_LAMBDA_NM:
        raise ValueError(f"the sky model is used above {_MIN_LAMBDA_NM:.0f} nm")
    return SkySpectrum(
        wavelength_edges=u.Quantity(wavelength_edges).to(u.nm),
        components={
            "moon": bin_to_channels(self._moon_lam, self.moonlight(conditions), edges_nm),
            "airglow": bin_to_channels(self._glow_lam, self.airglow(conditions), edges_nm),
            "zodiacal": bin_to_channels(self._zodi_lam, self.zodiacal(conditions), edges_nm),
            "starlight": bin_to_channels(self._zodi_lam, self.starlight(conditions), edges_nm),
        },
    )

    def airmass_from_altitude(altitude: u.Quantity) -> np.ndarray:
    """Relative airmass from apparent altitude (Kasten & Young 1989).

    ``X = 1 / [sin h + 0.50572 (h + 6.07995 deg)^-1.6364]`` (``h`` in
    degrees; Kasten & Young 1989, Appl. Opt. 28, 4735). Finite all the way to
    the horizon (X ~ 38), unlike ``sec z``. Returns ``inf`` for targets below
    the horizon (``h <= 0``).
    """
    h = np.atleast_1d(u.Quantity(altitude).to_value(u.deg)).astype(float)
    x = np.full(h.shape, np.inf)
    up = h > 0.0
    x[up] = 1.0 / (np.sin(np.radians(h[up])) + 0.50572 * (h[up] + 6.07995) ** -1.6364)
    return x

    def ecliptic_from_sun(coord: SkyCoord, times: Time) -> tuple[u.Quantity, u.Quantity]:
    """Target ecliptic longitude *relative to the Sun* (wrapped to +/-180 deg)
    and ecliptic latitude, as the zodiacal-light model needs.

    Geocentric true-of-date ecliptic. (SkyCalc's own almanac differed from this
    by ~4 deg in longitude for one tested target; see ``kino.observing.skymodel``.)
    """
    frame = GeocentricTrueEcliptic(obstime=times)
    target = coord.transform_to(frame)
    sun = get_sun(times).transform_to(frame)
    lon = ((target.lon - sun.lon + 180 * u.deg) % (360 * u.deg)) - 180 * u.deg
    return lon.to(u.deg), target.lat.to(u.deg)

    def moon_state(site: Site, times: Time, coord: SkyCoord | None = None) -> MoonState:
    """Moon illumination, phase angle, altitude and (optionally) separation.

    The phase angle uses the Sun-Moon elongation and the true distances
    (``i = atan2(R_sun sin e, R_moon - R_sun cos e)``); accurate to well under
    a degree, which is far finer than sky-brightness models need.
    """
    # geocentric Sun and Moon for the phase (both in the same frame); the
    # topocentric Moon for altitude and separation from the target
    moon_geo = get_body("moon", times)
    moon = get_body("moon", times, location=site.location)
    sun = get_sun(times)
    elong = sun.separation(moon_geo)
    r_sun, r_moon = sun.distance.to(u.au), moon_geo.distance.to(u.au)
    phase = np.arctan2(r_sun * np.sin(elong), r_moon - r_sun * np.cos(elong))
    altaz = moon.transform_to(AltAz(obstime=times, location=site.location))
    sep = None
    if coord is not None:
        # ``moon`` is in the site's topocentric GCRS frame and ``coord`` (a distant
        # target) is transformed into it, so this is the topocentric separation --
        # the one that matters for scattered moonlight (lunar parallax is ~1 deg).
        # astropy warns that the result can depend on the transform direction; for a
        # target at infinite distance it does not.
        with warnings.catch_warnings():
            warnings.simplefilter("ignore", NonRotationTransformationWarning)
            sep = moon.separation(coord).to(u.deg)
    return MoonState(
        illumination=(1.0 + np.cos(phase)) / 2.0,
        phase_angle=phase.to(u.deg),
        altitude=altaz.alt.to(u.deg),
        separation=sep,
    )

    seeing_at(seeing_ref: u.Quantity, airmass: float | np.ndarray, wavelength: u.Quantity, *, reference_wavelength: u.Quantity = REFERENCE_WAVELENGTH) -> u.Quantity ¶

    Effective seeing FWHM at an airmass and wavelength.

    Standard Kolmogorov scaling from a zenith seeing quoted at reference_wavelength (conventionally 500 nm): the Fried parameter goes as r0 ~ lambda^(6/5) X^(-3/5) and the FWHM as lambda / r0, so

    FWHM = FWHM_ref * X^(3/5) * (lambda / lambda_ref)^(-1/5).

    An approximation (it ignores the outer scale, which lowers the FWHM at long wavelengths, and dome/instrument seeing); use it to shape a seeing curve, not to predict a single image to better than ~10%.

    def seeing_at(
    seeing_ref: u.Quantity,
    airmass: float | np.ndarray,
    wavelength: u.Quantity,
    *,
    reference_wavelength: u.Quantity = REFERENCE_WAVELENGTH,
) -> u.Quantity:
    """Effective seeing FWHM at an airmass and wavelength.

    Standard Kolmogorov scaling from a zenith seeing quoted at
    ``reference_wavelength`` (conventionally 500 nm): the Fried parameter
    goes as ``r0 ~ lambda^(6/5) X^(-3/5)`` and the FWHM as ``lambda / r0``, so

        FWHM = FWHM_ref * X^(3/5) * (lambda / lambda_ref)^(-1/5).

    An approximation (it ignores the outer scale, which lowers the FWHM at
    long wavelengths, and dome/instrument seeing); use it to *shape* a
    seeing curve, not to predict a single image to better than ~10%.
    """
    x = np.asarray(airmass, dtype=float)
    if np.any(x < 1.0):
        raise ValueError("airmass must be >= 1")
    if u.Quantity(seeing_ref).value <= 0:
        raise ValueError("seeing_ref must be positive")
    scale = x**0.6 * (u.Quantity(wavelength) / reference_wavelength).decompose().value ** -0.2
    return u.Quantity(seeing_ref).to(u.arcsec) * scale

    def target_track(site: Site, coord: SkyCoord, times: Time) -> TargetTrack:
    """Altitude, airmass and parallactic angle of ``coord`` at ``times``."""
    altaz = coord.transform_to(AltAz(obstime=times, location=site.location))
    # hour angle needs the target's RA *of date* (apparent sidereal time is
    # referred to the true equinox of date); J2000 RA would be off by the
    # accumulated precession (~0.3 deg after 25 years).
    of_date = coord.transform_to(TETE(obstime=times))
    lst = times.sidereal_time("apparent", longitude=site.location.lon)
    hour_angle = (lst - of_date.ra).wrap_at(180 * u.deg)
    lat, dec = site.location.lat, of_date.dec
    q = np.arctan2(
        np.sin(hour_angle),
        np.tan(lat) * np.cos(dec) - np.sin(dec) * np.cos(hour_angle),
    )
    return TargetTrack(
        altitude=altaz.alt.to(u.deg),
        azimuth=altaz.az.to(u.deg),
        airmass=airmass_from_altitude(altaz.alt),
        parallactic_angle=q.to(u.deg),
    )

    def extinction_coefficient(wavelength: u.Quantity) -> np.ndarray:
    """CTIO extinction coefficient ``k`` (mag per airmass) at ``wavelength``.

    Linear interpolation in wavelength. Raises ``ValueError`` outside
    3200-10870 A rather than extrapolating.
    """
    wave = np.atleast_1d(u.Quantity(wavelength).to_value(u.angstrom)).astype(float)
    lo, hi = CTIO_WAVELENGTH_ANGSTROM[[0, -1]]
    tol = 1e-6  # Angstrom: unit-conversion round-off at the table edges
    if np.any(wave < lo - tol) or np.any(wave > hi + tol):
        raise ValueError(f"wavelength outside the tabulated CTIO range {lo:.0f}-{hi:.0f} A")
    return np.interp(wave, CTIO_WAVELENGTH_ANGSTROM, CTIO_EXTINCTION_MAG_PER_AIRMASS)

    def nir_extinction_coefficient(wavelength: u.Quantity) -> np.ndarray:
    """Broadband CTIO J/H/Ks extinction coefficient (mag per airmass) at ``wavelength``.

    No interpolation between bands (see the module docstring): ``wavelength``
    must fall inside one band's own isophotal-bandwidth window. Raises
    ``ValueError`` otherwise, naming the covered windows.
    """
    wave_um = np.atleast_1d(u.Quantity(wavelength).to_value(u.um)).astype(float)
    out = np.full(wave_um.shape, np.nan)
    covered = np.zeros(wave_um.shape, dtype=bool)
    for band in NIR_EXTINCTION_BANDS:
        lo, hi = band.center_um - band.halfwidth_um, band.center_um + band.halfwidth_um
        inside = (wave_um >= lo) & (wave_um <= hi)
        out[inside] = band.mag_per_airmass
        covered |= inside
    if not covered.all():
        bad = np.round(wave_um[~covered], 4)
        windows = ", ".join(
            f"{b.name} {b.center_um - b.halfwidth_um:.3f}-{b.center_um + b.halfwidth_um:.3f} um"
            for b in NIR_EXTINCTION_BANDS
        )
        raise ValueError(
            f"wavelength(s) {bad} um fall outside every covered NIR band ({windows}) -- "
            "no measured broadband extinction there (see the module docstring)"
        )
    return out

    site: Site,
    date: str | Time,
    *,
    sun_altitude_max: u.Quantity = -18.0 * u.deg,
    step: u.Quantity = 5.0 * u.min,
) -> Night:
    """The dark window of the night that *begins* on local evening ``date``.

    ``date`` is a calendar date (``"2026-10-03"``): the evening of that date
    at the site, running into the next morning. The search starts at the
    site's mean solar noon on that date (12h UT minus the longitude in hours)
    and takes the first contiguous stretch with the Sun below
    ``sun_altitude_max`` -- so no timezone or daylight-saving handling is
    needed, and the result is unambiguous across the UT date boundary.

    Parameters
    ----------
    sun_altitude_max : Quantity (angle), optional
        Default -18 deg (astronomical twilight); use -12 deg for nautical.
    step : Quantity (time), optional
        Grid spacing.
    """
    day = Time(date, scale="utc")
    noon = Time(int(day.mjd) + 0.5, format="mjd", scale="utc") - site.longitude.hour * u.hour
    n = int(round((24 * u.hour / step).decompose().value))
    times = noon + np.arange(n) * step

    sun_alt = get_sun(times).transform_to(AltAz(obstime=times, location=site.location)).alt
    dark = sun_alt < sun_altitude_max
    if not dark.any():
        raise ValueError(
            f"the Sun never drops below {sun_altitude_max} at {site.name} on {day.iso[:10]}"
        )

    first = int(np.argmax(dark))
    stretch = dark[first:]
    length = int(np.argmin(stretch)) if not stretch.all() else len(stretch)
    sel = slice(first, first + length)
    return Night(
        times=times[sel],
        sun_altitude=sun_alt[sel].to(u.deg),
        start=times[sel][0],
        end=times[sel][-1],
    )

    def moonlight_brightness(
    phase_angle: u.Quantity,
    separation: u.Quantity,
    moon_zenith_distance: u.Quantity,
    zenith_distance: u.Quantity,
    k: float,
) -> np.ndarray:
    """Scattered moonlight in the V band, nanoLamberts (K&S91 Section 5).

    ::

        I*   = 10^(-0.4 (3.84 + 0.026|a| + 4e-9 a^4))     lunar illuminance
        f(r) = 10^5.36 (1.06 + cos^2 r) + 10^(6.15 - r/40)    r >= 10 deg
        f(r) = 10^5.36 (1.06 + cos^2 r) + 6.2e7 / r^2         r <  10 deg
        B    = f(r) I* 10^(-0.4 k X(Zm)) [1 - 10^(-0.4 k X(Z))]

    with ``a`` the lunar phase angle (0 = full), ``r`` the Moon-target
    separation, ``Zm``/``Z`` the zenith distances of the Moon and the target.
    The ``r < 10 deg`` branch is the aureole term (K&S91 eq. 19, Schaefer 1991)
    replacing the Mie term; the paper's summary formula is stated for
    r > 10 deg only. Returns 0 when the Moon is below the horizon.

    ``phase_angle``, ``separation`` and the zenith distances broadcast
    together; ``k`` is the V-band extinction (mag per airmass).
    """
    alpha = np.abs(np.asarray(u.Quantity(phase_angle).to_value(u.deg), dtype=float))
    rho = np.asarray(u.Quantity(separation).to_value(u.deg), dtype=float)
    zm = u.Quantity(moon_zenith_distance)
    z = u.Quantity(zenith_distance)

    i_star = 10.0 ** (-0.4 * (3.84 + 0.026 * alpha + 4e-9 * alpha**4))
    cos2 = np.cos(np.radians(rho)) ** 2
    rayleigh = 10.0**5.36 * (1.06 + cos2)
    with np.errstate(divide="ignore"):
        mie = np.where(rho >= 10.0, 10.0 ** (6.15 - rho / 40.0), 6.2e7 / np.maximum(rho, 1e-6) ** 2)
    f = rayleigh + mie

    x_moon = _night_glow_airmass(zm)
    x_sky = _night_glow_airmass(z)
    b = f * i_star * 10.0 ** (-0.4 * k * x_moon) * (1.0 - 10.0 ** (-0.4 * k * x_sky))
    return np.where(np.asarray(zm.to_value(u.deg)) < 90.0, b, 0.0)

    def sky_electron_rate(
    sky: SkySpectrum,

    kino.observing.skymodel ¶

    Night-sky radiance spectra from a local library of ESO SkyCalc output.

    SkyLibrary interpolates, offline, spectra of the four sky components that matter at optical wavelengths -- scattered moonlight, airglow (emission lines plus residual continuum), zodiacal light and scattered starlight -- computed with ESO's Cerro Paranal Advanced Sky Model (Noll et al. 2012, A&A 543, A92; Jones et al. 2013, A&A 560, A91). The library is built once from the SkyCalc service by scripts/build_skycalc_library.py (network, ~10 minutes) and kept in data/skycalc/ (git-ignored), so batch runs are offline and reproducible.

    Radiance is in photons / s / m^2 / micron / arcsec^2 at the ground (it already includes the sky's own extinction); wavelengths are vacuum, in nm.

    What was checked against independent references (V band, see the tests and CLAUDE.md): the dark zenith sky is 21.9 mag/arcsec^2 at 100 s.f.u. against the CTIO mean 21.79 +/- 0.22 (Krisciunas et al. 2007), and the moonlight brightening agrees with Krisciunas & Schaefer (1991) to 0.1-0.4 mag across four geometries.

    Approximations and known limits -- read before trusting a number:

    • the model is for Cerro Paranal (2640 m); Cerro Pachon (SOAR/Gemini South) has more water vapour (irrelevant below ~1 micron), different aerosols, and La Serena's light pollution at low elevation, none of which is modelled;
    • airglow is averaged over seasons and the whole night, although it varies by tens of percent within a night; it scales with the 10.7 cm solar flux, which the caller must supply for the epoch (there is no default you can trust);
    • moonlight is interpolated (log-linear) on a grid in phase, target-Moon separation, Moon zenith distance and airmass, capped at airmass 2 and at a Moon zenith distance of 80 deg with a linear taper to zero at the horizon;
    • zodiacal light and starlight are stored at airmass 1 and scaled by the CTIO extinction curve; the ecliptic longitude is measured relative to the Sun and differs by ~4 deg from SkyCalc's own almanac for one tested target, which matters little far from the Sun;
    • wavelengths are vacuum -- convert if you compare sky lines with air wavelengths.

    @classmethod
def load(cls, path: str | Path = DEFAULT_LIBRARY_PATH) -> SkyLibrary:
    """Load the library built by ``scripts/build_skycalc_library.py``."""
    path = Path(path)
    try:
        moon = dict(np.load(path / "moon.npz"))
        airglow = dict(np.load(path / "airglow.npz"))
        zodi = dict(np.load(path / "zodi.npz"))
    except FileNotFoundError as exc:
        raise FileNotFoundError(
            f"no sky library in {path}/ -- build it once (needs network and "
            "`pip install skycalc-cli`): python scripts/build_skycalc_library.py"
        ) from exc
    meta = {}
    if (path / "meta.json").exists():
        import json

        meta = json.loads((path / "meta.json").read_text())
    return cls(moon, airglow, zodi, meta)

    def spectrum(self, conditions: SkyConditions, wavelength_edges: u.Quantity) -> SkySpectrum:
    """Sky components averaged over the channels bounded by ``wavelength_edges``.

    ``wavelength_edges`` are *vacuum* wavelengths, shape ``(n + 1,)``.
    """
    if conditions.airmass < 1.0:
        raise ValueError("airmass must be >= 1")
    edges_nm = u.Quantity(wavelength_edges).to_value(u.nm)
    if edges_nm[0] < _MIN_LAMBDA_NM:
        raise ValueError(f"the sky model is used above {_MIN_LAMBDA_NM:.0f} nm")
    return SkySpectrum(
        wavelength_edges=u.Quantity(wavelength_edges).to(u.nm),
        components={
            "moon": bin_to_channels(self._moon_lam, self.moonlight(conditions), edges_nm),
            "airglow": bin_to_channels(self._glow_lam, self.airglow(conditions), edges_nm),
            "zodiacal": bin_to_channels(self._zodi_lam, self.zodiacal(conditions), edges_nm),
            "starlight": bin_to_channels(self._zodi_lam, self.starlight(conditions), edges_nm),
        },
    )

    def bin_to_channels(lam_nm: np.ndarray, radiance: np.ndarray, edges_nm: np.ndarray) -> np.ndarray:
    """Mean of ``radiance(lam)`` over each channel ``[edges[i], edges[i+1]]``.

    Integrates the piecewise-linear spectrum exactly via its cumulative
    integral, so narrow sky lines contribute their true flux to a wide channel
    instead of being sampled or missed. Channels partly outside the library
    range are averaged over the covered part; fully outside raises.
    """
    edges = np.asarray(edges_nm, dtype=float)
    if edges[0] < lam_nm[0] or edges[-1] > lam_nm[-1]:
        raise ValueError(
            f"channels {edges[0]:.1f}-{edges[-1]:.1f} nm extend outside the sky library "
            f"({lam_nm[0]:.1f}-{lam_nm[-1]:.1f} nm)"
        )
    cum = np.concatenate([[0.0], np.cumsum(0.5 * (radiance[1:] + radiance[:-1]) * np.diff(lam_nm))])
    at_edges = np.interp(edges, lam_nm, cum)
    return np.diff(at_edges) / np.diff(edges)

    def sky_electron_rate(
    sky: SkySpectrum,
    *,
    pixel_area: u.Quantity,
    effective_area: u.Quantity,
    throughput,
) -> u.Quantity:
    """Detected sky electrons per second per pixel per channel.

    ``rate = radiance * channel_width * pixel_area * effective_area * throughput``
    -- the same collecting area and end-to-end throughput
    :func:`kino.instruments.base.photon_flux_to_electrons` applies to the
    source, so source and sky are on one footing. ``throughput`` is a
    :class:`~kino.instruments.base.Throughput`, evaluated at the channel
    centres. Suitable as ``IFU.sky_electron_rate`` (which multiplies by the
    total exposure time itself).
    """
    radiance = sky.total * (u.photon / u.s / u.m**2 / u.um / u.arcsec**2)
    photons = radiance * sky.channel_width.to(u.um) * pixel_area.to(u.arcsec**2) * effective_area
    rate = photons.to(u.photon / u.s) * throughput(sky.wavelength)
    return rate.value * (u.electron / u.s)
